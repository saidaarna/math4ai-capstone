% Slide 2: The Central Question
\begin{frame}{The Central Question}
    \begin{block}{Our Objective}
        \textit{When does a one-hidden-layer nonlinear classifier genuinely improve on a linear decision rule, and when is additional model complexity unnecessary?}
    \end{block}
    
    \vspace{0.5em}
    \textbf{Models Compared:}
    \begin{itemize}
        \item \textbf{Softmax Regression (SR):} Linear score function, affine decision boundaries.
        \item \textbf{1-Hidden-Layer Neural Network (NN):} Non-linear representation via $\tanh$ activation, followed by linear classification.
    \end{itemize}
    
    \vspace{0.5em}
    All components (forward pass, backpropagation, and optimization) were implemented entirely from scratch in NumPy.
\end{frame}

% Slide 3: Ensuring a Fair Comparison
\begin{frame}{Ensuring a Fair Comparison}
    Because the models differ in structure and parameterization, our comparison must be mathematically fair.
    
    \vspace{1em}
    \textbf{Our Experimental Contract:}
    \begin{itemize}
        \item \textbf{Identical Splits:} Fixed Train / Validation / Test indices across all runs.
        \item \textbf{Identical Preprocessing:} Digits used exactly as provided (scaled to $[0,1]$), with no task-specific feature engineering.
        \item \textbf{Strict Model Selection:} Settings and checkpoints chosen \textit{only} based on Validation Cross-Entropy.
        \item \textbf{Evaluation Metrics:} Unregularized Test Cross-Entropy (measuring predictive distribution quality) and Test Accuracy.
        \item \textbf{Statistical Rigor:} Final evaluations on the digits benchmark use 5 repeated random seeds.
    \end{itemize}
\end{frame}

% Slide 7: What Repeated Seeds Add
\begin{frame}{What Repeated-Seed Evaluation Adds}
    A single run can be lucky or unlucky due to random initialization. Re-running with 5 seeds provides statistical confidence.
    
    \vspace{1em}
    \begin{table}
        \centering
        \begin{tabular}{lcc}
            \toprule
            \textbf{Model} & \textbf{Test Accuracy (95\% CI)} & \textbf{Test CE (95\% CI)} \\
            \midrule
            Softmax Regression  & $0.9375 \pm 0.0024$ & $0.2683 \pm 0.0027$ \\
            1-Hidden-Layer NN   & $0.9511 \pm 0.0000$ & $0.1671 \pm 0.0077$ \\
            \bottomrule
        \end{tabular}
    \end{table}
    
    \vspace{1em}
    \textbf{Takeaways:}
    \begin{itemize}
        \item Non-overlapping confidence intervals confirm the NN's advantage is statistically robust.
        \item SR is highly stable but bounded by its linear representational ceiling.
        \item The NN has higher variance in its predictive distributions (Cross-Entropy).
    \end{itemize}
\end{frame}

% Slide 8: Failure Case Analysis
\begin{frame}{Failure Case Analysis: Optimization Limits}
    \textbf{Experiment:} Moons task, default constraints (200 epochs, $\eta=0.05$).
    
    \begin{center}
        \includegraphics[width=0.9\textwidth]{figures/capacity_ablation_default_combined.png}
    \end{center}
    
    \vspace{-0.5em}
    \begin{itemize}
        \item The boundary remains near-linear across \textit{all} widths, even with excessive capacity ($m=32$).
        \item \textbf{Why?} The default settings are too slow to escape the initial plateau within the strict 200-epoch budget.
        \item \textbf{Lesson:} Highly complex models are useless without the proper mathematical optimization to train them.
    \end{itemize}
\end{frame}

% Slide 9: Capacity Ablation (Optimized)
\begin{frame}{Capacity Ablation: Complexity is not always better}
    \textbf{Experiment:} Moons task, optimized training (1000 epochs, $\eta=0.1$).
    
    \begin{center}
        \includegraphics[width=0.9\textwidth]{figures/capacity_ablation_combined.png}
    \end{center}
    
    \vspace{-0.5em}
    \begin{itemize}
        \item Given proper optimization, validation precision reaches $0.9875$ identically across all widths.
        \item \textbf{Key Insight:} \textbf{A highly complex model is not always better.} Introducing massive capacity ($m=32$) provided no advantage over $m=2$; the extremely lightweight network was fully sufficient.
    \end{itemize}
\end{frame}

% Slide 11: Limitations
\begin{frame}{Limitations of Our Study}
    \begin{itemize}
        \item \textbf{Dataset Scale:} The digits dataset contains only 1,797 samples. Our findings may not seamlessly extrapolate to large-scale vision or language regimes.
        \item \textbf{Architectural Depth:} We analyzed exactly \textit{one} hidden layer. We make no claims about the optimization dynamics or representational power of deeper networks.
        \item \textbf{Low-Dimensional Geometry:} The clear visual intuitions from the 2D Gaussian and Moons tasks do not perfectly mirror the geometric realities of high-dimensional feature spaces.
    \end{itemize}
\end{frame}

% Slide 12: Summary
\begin{frame}{Summary}
    \begin{itemize}
        \item \textbf{When is linear sufficient?} When the data geometry dictates an affine separator (e.g., Gaussian task).
        \item \textbf{What does the hidden layer add?} It warps the input space, allowing a linear output layer to solve nonlinearly separable problems (e.g., Moons, Digits).
        \item \textbf{Is complexity always justified?} No. If structural capacity is too low (our failure case), or the task geometry is too simple, the extra complexity adds nothing.
        \item \textbf{Overall:} The One-Hidden-Layer NN outperformed Softmax Regression significantly on the digits benchmark in both raw accuracy and calibrated reliability.
    \end{itemize}
